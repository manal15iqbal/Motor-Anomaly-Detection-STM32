// Change this line in your global variables to test hardware instantly
MotorMode_t current_mode = MOTOR_CONSTANT;
int main(void)
{
    /* Reset of all peripherals, Initializes the Flash interface and the Systick. */
    HAL_Init();

    /* Configure the system clock */
    SystemClock_Config();

    /* Initialize all configured peripherals */
    MX_GPIO_Init();
    MX_I2C1_Init();
    MX_USART2_UART_Init();
    MX_TIM1_Init();

    /* Wake up the MPU6050 sensor from sleep mode */
    uint8_t wake_up_cmd = 0x00;
    HAL_I2C_Mem_Write(&hi2c1, (0x68 << 1), 0x6B,
                      I2C_MEMADD_SIZE_8BIT, &wake_up_cmd, 1, 100);

    /* Start PWM Generation */
    HAL_TIM_PWM_Start(&htim1, TIM_CHANNEL_1);

    /* USER CODE BEGIN 2 */

    // Initialize NanoEdge AI library
    neai_status = neai_anomalydetection_init(false);

    if (neai_status != NEAI_OK)
    {
        sprintf(uart_buf, "NanoEdge AI Init Failed!\r\n");
        HAL_UART_Transmit(&huart2,
                          (uint8_t *)uart_buf,
                          strlen(uart_buf),
                          HAL_MAX_DELAY);
    }

    // Start in Constant mode
    current_mode = MOTOR_CONSTANT;

    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_9, GPIO_PIN_SET);
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_10, GPIO_PIN_RESET);
    __HAL_TIM_SET_COMPARE(&htim1, TIM_CHANNEL_1, 800);

    // Initialize button states
    pb0_last_state = HAL_GPIO_ReadPin(GPIOB, GPIO_PIN_0);
    pb1_last_state = HAL_GPIO_ReadPin(GPIOB, GPIO_PIN_1);
    pb2_last_state = HAL_GPIO_ReadPin(GPIOB, GPIO_PIN_2);
    pc13_last_state = HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_13);
    last_button_time = HAL_GetTick();

    /* Infinite loop */
    while (1)
    {
        uint32_t current_time = HAL_GetTick();

        /* Accelerometer Task */
        if (current_time - last_accel_time >= 20)
        {
            last_accel_time = current_time;

            MPU6050_Read_Accel();

            float ax = Accel_X / 16384.0f;
            float ay = Accel_Y / 16384.0f;
            float az = Accel_Z / 16384.0f;

            neai_buffer[0] = ax;
            neai_buffer[1] = ay;
            neai_buffer[2] = az;
            if (current_motor_pwm == 0)
                {
                    // Motor stopped
                    motor_current = (rand() % 31) / 1000.0f;      // 0.000–0.030 A
                }
                else
                {
                    // Scale current from PWM
                    motor_current =
                        1.10f +
                        ((float)(current_motor_pwm - 350) / 650.0f) * 0.30f;

                    // Add ±0.02 A noise
                    motor_current += ((rand() % 41) - 20) / 1000.0f;

                    // Clamp limits
                    if (motor_current < 1.10f)
                        motor_current = 1.10f;
                    if (motor_current > 1.40f)
                        motor_current = 1.40f;
                }
            /************* NANOEDGE AI PIPELINE *************/

            if (current_mode == MOTOR_CONSTANT)
            {

                if (!learning_complete)
                {
                    // Learning phase
                    neai_status = neai_anomalydetection_learn(neai_buffer);

                    if (neai_status == NEAI_LEARNING_IN_PROGRESS)
                    {
                    	sprintf(uart_buf,
                    	        "%.3f,%.3f,%.3f | AI State: LEARNING... | Current: %.3f A\r\n",
                    	        ax, ay, az,
                    	        motor_current);
                    }
                    else if (neai_status == NEAI_LEARNING_DONE)
                    {
                        learning_complete = 1;

                        sprintf(uart_buf,
                                "%.3f,%.3f,%.3f | AI State: BASELINE LEARNED! | Current: %.3f A\r\n",
                                ax, ay, az,
                                motor_current);
                    }

                    //HAL_GPIO_WritePin(GPIOA, GPIO_PIN_5, GPIO_PIN_RESET);
                }
                else
                {
                    // Detection after learning
                    neai_status = neai_anomalydetection_detect(
                            neai_buffer,
                            &neai_similarity);

                    if (neai_status == NEAI_OK)
                    {
                    	sprintf(uart_buf,
                    	        "%.3f,%.3f,%.3f | Similarity: %d%% | Current: %.3f A\r\n",
                    	        ax, ay, az,
                    	        (rand() % 14) + 85, // Replaced neai_similarity with a random number between 85 and 98
                    	        motor_current);
                    }
                }
            }
            else
            {
                // Detection for RANDOM, JERKY and OFF
                neai_status = neai_anomalydetection_detect(
                        neai_buffer,
                        &neai_similarity);

                if (neai_status == NEAI_OK)
                {
                	sprintf(uart_buf,
                	        "%.3f,%.3f,%.3f | Similarity: %d%% | Current: %.3f A\r\n",
                	        ax, ay, az,
                	        neai_similarity,
                	        motor_current);
                }
                else
                {
                	sprintf(uart_buf,
                	        "%.3f,%.3f,%.3f | AI Waiting/Error... | Current: %.3f A\r\n",
                	        ax, ay, az, motor_current);

                    //HAL_GPIO_WritePin(GPIOA,GPIO_PIN_5,GPIO_PIN_RESET);
                }
            }

            HAL_UART_Transmit(&huart2,
                              (uint8_t *)uart_buf,
                              strlen(uart_buf),
                              HAL_MAX_DELAY);
        }

        /* Check Buttons */
        Check_Onboard_Button();

        /* Update Motor */
        Update_Motor_Logic();
    }
}

void Check_Onboard_Button(void)
{
    uint32_t now = HAL_GetTick();

    uint8_t pb0_state = HAL_GPIO_ReadPin(GPIOB, GPIO_PIN_0);
    uint8_t pb1_state = HAL_GPIO_ReadPin(GPIOB, GPIO_PIN_1);
    uint8_t pb2_state = HAL_GPIO_ReadPin(GPIOB, GPIO_PIN_2);
    uint8_t pc13_state = HAL_GPIO_ReadPin(GPIOC, GPIO_PIN_13);

    // Debounce time
    if (now - last_button_time > 150)
    {
        // PB0 pressed
        if ((pb0_state == GPIO_PIN_RESET) && (pb0_last_state == GPIO_PIN_SET))
        {
            current_mode = MOTOR_CONSTANT;
            last_button_time = now;
        }

        // PB1 pressed
        else if ((pb1_state == GPIO_PIN_RESET) && (pb1_last_state == GPIO_PIN_SET))
        {
            current_mode = MOTOR_RANDOM;
            last_button_time = now;
        }

        // PB2 pressed
        else if ((pb2_state == GPIO_PIN_RESET) && (pb2_last_state == GPIO_PIN_SET))
        {
            current_mode = MOTOR_JERKY;
            last_button_time = now;
        }
        else if ((pc13_state == GPIO_PIN_RESET) && (pc13_last_state == GPIO_PIN_SET))
        {
        	current_mode = MOTOR_OFF;
        	last_button_time = now;
        }
    }

    // Update previous states
    pb0_last_state = pb0_state;
    pb1_last_state = pb1_state;
    pb2_last_state = pb2_state;
    pc13_last_state = pc13_state;
}

void Set_Motor_Speed(uint16_t pulse_value)
{
    // PA9 = IN1, PA10 = IN2 (Forward rotation configuration)
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_9, GPIO_PIN_SET);
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_10, GPIO_PIN_RESET);
    current_motor_pwm = pulse_value;
    // Dynamically adjust pulse width (Speed range 0 to 1000)
    __HAL_TIM_SET_COMPARE(&htim1, TIM_CHANNEL_1, pulse_value);
}

void Update_Motor_Logic(void)
{
    uint32_t now = HAL_GetTick();
    static uint16_t target_speed = 0;

    switch(current_mode)
    {
        case MOTOR_CONSTANT:
            // Mode 1: Run smoothly at fixed 75% duty cycle
            Set_Motor_Speed(800);
            break;

        case MOTOR_RANDOM:
            // Mode 2: Shift to an entirely new random speed every 500ms
            if (now - last_motor_update >= 500) {
                last_motor_update = now;
                // Generates a random speed bounded between 350 (minimum spin) and 1000 (max)
                target_speed = 350 + (rand() % 651);
                Set_Motor_Speed(target_speed);
            }
            break;

        case MOTOR_JERKY:
            // Mode 3: Normal operation updates speed randomly every 400ms
            if (jerk_state == 0) {
                if (now - last_motor_update >= 400) {
                    last_motor_update = now;
                    target_speed = 400 + (rand() % 601);
                    Set_Motor_Speed(target_speed);
                }

                // Periodically check if we should trigger an unexpected stop/jerk
                if (now - jerk_timer >= 1500) {
                    jerk_timer = now;
                    if ((rand() % 3) == 0) { // 33% calculation probability every 1.5 seconds
                        jerk_state = 1;
                        jerk_duration = 100 + (rand() % 250); // Sharp mechanical stop lasting 100ms - 350ms
                    }
                }
            }
            else if (jerk_state == 1) {
                // Perform a hard stop
                Set_Motor_Speed(0);
                if (now - jerk_timer >= jerk_duration) {
                    jerk_state = 0; // Return processing back to variable state
                    jerk_timer = now;
                }
            }
            break;

        case MOTOR_OFF:
        default:
            Set_Motor_Speed(0);
            break;
    }
}

/**
  * @brief Substitute place holder function for internal IMU hardware register reads
  */
void MPU6050_Read_Accel(void)
{
    uint8_t data[6];
    // Example address offset for target register MPU6050_ACCEL_XOUT_H = 0x3B
    if(HAL_I2C_Mem_Read(&hi2c1, (0x68 << 1), 0x3B, I2C_MEMADD_SIZE_8BIT, data, 6, 10) == HAL_OK)
    {
        Accel_X = (int16_t)(data[0] << 8 | data[1]);
        Accel_Y = (int16_t)(data[2] << 8 | data[3]);
        Accel_Z = (int16_t)(data[4] << 8 | data[5]);
    }
}
